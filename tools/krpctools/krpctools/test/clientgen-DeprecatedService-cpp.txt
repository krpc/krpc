namespace encoder {

  inline std::string encode(const services::DeprecatedService::ServiceEnum& value) {
    return krpc::encoder::encode(static_cast<int32_t>(value));
  }

  // A structure is sent as the values of its fields in order, which is the same encoding as
  // a tuple of those values. See the note on the collection encoders for why the message it
  // is sent in is kept between calls.
  inline std::string encode(const services::DeprecatedService::ServiceStruct& value) {
    static thread_local krpc::schema::Tuple structMessage;
    structMessage.Clear();
    structMessage.add_items(encode_item(value.field));
    return krpc::encoder::encode(structMessage);
  }

}  // namespace encoder

namespace decoder {

  inline void decode(services::DeprecatedService::ServiceEnum& value, const std::string& data, ::krpc::Connection* connection) {
    int32_t x;
    decode(x, data, connection);
    value = static_cast<services::DeprecatedService::ServiceEnum>(x);
  }

  // Fields are only ever appended to a structure, so a value from a newer server may carry
  // more items than this client knows about, and those are ignored.
  inline void decode(services::DeprecatedService::ServiceStruct& value, const std::string& data, ::krpc::Connection* connection) {
    static thread_local krpc::schema::Tuple structMessage;
    if (!structMessage.ParseFromString(data)) throw EncodingError("Failed to decode message");
    if (structMessage.items_size() < 1)
      throw EncodingError("Value for ServiceStruct does not have all of its fields");
    decode_item(value.field, structMessage.items(0), connection);
  }

}  // namespace decoder

inline std::size_t hash_value(const services::DeprecatedService::ServiceStruct& value) {
  std::size_t seed = 0;
  hash_combine(&seed, hash_value(value.field));
  return seed;
}

// A collection encoder or decoder calls encode and decode on the type it contains
// without qualifying them. The ones above are declared after those templates, so
// argument dependent lookup is what reaches them. It searches the namespace that
// encloses the type, which is this one. These name the same functions here, as a
// second function with the same signature is ambiguous to a compiler that finds both.
namespace services {
  using krpc::encoder::encode;
  using krpc::decoder::decode;
  using krpc::hash_value;
}  // namespace services
