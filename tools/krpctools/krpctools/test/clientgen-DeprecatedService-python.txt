    """
    Deprecated. Use TestService instead.

"""

@deprecated("Use \"TestService.TestClass\" instead.")
class ReasonedClass(DeclaredObject, metaclass=DeclaredClass):
    """
    Deprecated. Use "TestService.TestClass" instead.

    A class deprecated with a reason of its own.
    """
    _service_name = "DeprecatedService"
    _class_name = "ReasonedClass"

    @staticmethod
    @deprecated("Use \"TestService.TestClass\" instead.")
    def static_method() -> str:
        """
        Deprecated. Use "TestService.TestClass" instead.

        A static method.
        """
        raise _declaration("DeprecatedService", "ReasonedClass")

# Aliases for the types the service declares as its attributes, as an annotation in the body of

if TYPE_CHECKING:
    _ReasonedClass = ReasonedClass
    _ServiceClass = ServiceClass
    _ServiceEnum = ServiceEnum
    _ServiceStruct = ServiceStruct
    _ServiceException = ServiceException


@deprecated("Use TestService instead.")
class DeprecatedService(metaclass=DeclaredClass):
    """
    A deprecated service, whose members are deprecated only through it, other than one class.
    """
    _service_name = "DeprecatedService"
    _class_name = None

    ReasonedClass: type[_ReasonedClass] = ReasonedClass
    ServiceClass: type[_ServiceClass] = ServiceClass
    ServiceEnum: type[_ServiceEnum] = ServiceEnum
    ServiceStruct: type[_ServiceStruct] = ServiceStruct
    ServiceException: type[_ServiceException] = ServiceException

    @property
    @deprecated("Use TestService instead.")
    def property(self) -> str:
        """
        Deprecated. Use TestService instead.

        A property.
        """
        raise _declaration("DeprecatedService", None)

    @property.setter
    @deprecated("Use TestService instead.")
    def property(self, value: str) -> None:
        raise _declaration("DeprecatedService", None)

    @deprecated("Use TestService instead.")
    def procedure(self) -> str:
        """
        Deprecated. Use TestService instead.

        A procedure.
        """
        raise _declaration("DeprecatedService", None)

    @deprecated("Use TestService instead.")
    def struct_procedure(self) -> ServiceStruct:
        """
        Deprecated. Use TestService instead.

        A procedure returning a structure.
        """
        raise _declaration("DeprecatedService", None)


class DeprecatedServiceClient(krpc.client.Client):
    """
    A client with the DeprecatedService service, as ``krpc.connect[DeprecatedServiceClient]()``
    connects. Combine it with the client types of other services in a subclass of them all.
    """

    _declares_service = "DeprecatedService"

    deprecated_service: DeprecatedService


_register(
    "DeprecatedService",
    ServiceEnum,
    ServiceStruct,
    ServiceException,
)
