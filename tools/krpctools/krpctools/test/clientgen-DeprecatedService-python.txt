    """
    Deprecated. Use TestService instead.

"""

@_deprecated("Use \"TestService.TestClass\" instead.")
class ReasonedClass(_DeclaredObject, metaclass=_DeclaredClass):
    """
    Deprecated. Use "TestService.TestClass" instead.

    A class deprecated with a reason of its own.
    """
    _service_name = "DeprecatedService"
    _class_name = "ReasonedClass"

    @_builtins.staticmethod
    @_deprecated("Use \"TestService.TestClass\" instead.")
    def static_method() -> _builtins.str:
        """
        Deprecated. Use "TestService.TestClass" instead.

        A static method.
        """
        raise _declaration("DeprecatedService", "ReasonedClass")

# Aliases for the types the service declares as its attributes, as an annotation in the body of
# the service would otherwise name the attribute itself
if TYPE_CHECKING:
    _type_ReasonedClass = ReasonedClass
    _type_ServiceClass = ServiceClass
    _type_ServiceEnum = ServiceEnum
    _type_ServiceStruct = ServiceStruct
    _type_ServiceException = ServiceException


@_deprecated("Use TestService instead.")
class DeprecatedService(metaclass=_DeclaredClass):
    """
    A deprecated service, whose members are deprecated only through it, other than one class.
    """
    _service_name = "DeprecatedService"
    _class_name = None

    ReasonedClass: type[_type_ReasonedClass] = ReasonedClass
    ServiceClass: type[_type_ServiceClass] = ServiceClass
    ServiceEnum: type[_type_ServiceEnum] = ServiceEnum
    ServiceStruct: type[_type_ServiceStruct] = ServiceStruct
    ServiceException: type[_type_ServiceException] = ServiceException

    @_builtins.property
    @_deprecated("Use TestService instead.")
    def property(self) -> _builtins.str:
        """
        Deprecated. Use TestService instead.

        A property.
        """
        raise _declaration("DeprecatedService", None)

    @property.setter
    @_deprecated("Use TestService instead.")
    def property(self, value: _builtins.str) -> None:
        raise _declaration("DeprecatedService", None)

    @_deprecated("Use TestService instead.")
    def procedure(self) -> _builtins.str:
        """
        Deprecated. Use TestService instead.

        A procedure.
        """
        raise _declaration("DeprecatedService", None)

    @_deprecated("Use TestService instead.")
    def struct_procedure(self) -> ServiceStruct:
        """
        Deprecated. Use TestService instead.

        A procedure returning a structure.
        """
        raise _declaration("DeprecatedService", None)


class DeprecatedServiceConnection(krpc.connection.Connection):
    """
    A connection with the DeprecatedService service, as ``krpc.connect[DeprecatedServiceConnection]()``
    connects. Combine it with the connection types of other services in a subclass of them all.
    """

    _declares_service = "DeprecatedService"

    deprecated_service: DeprecatedService


_register(
    "DeprecatedService",
    ServiceEnum,
    ServiceStruct,
    ServiceException,
)
