static inline bool krpc_encode_callback_items_DeprecatedService_ServiceStruct(
  pb_ostream_t * stream, const pb_field_t * field, void * const * arg) {
  const krpc_DeprecatedService_ServiceStruct_t * value = (const krpc_DeprecatedService_ServiceStruct_t*)(*arg);
  {
    if (!pb_encode_tag_for_field(stream, field))
      KRPC_CALLBACK_RETURN_ERROR("encoding tag for tuple item");
    size_t size;
    KRPC_CALLBACK_RETURN_ON_ERROR(krpc_encode_size_int32(&size, value->field));
    if (!pb_encode_varint(stream, size))
      KRPC_CALLBACK_RETURN_ERROR("encoding size for tuple item");
    KRPC_CALLBACK_RETURN_ON_ERROR(krpc_encode_int32(stream, value->field));
  }
  return true;
}

static inline krpc_error_t krpc_encode_DeprecatedService_ServiceStruct(
  pb_ostream_t * stream, const krpc_DeprecatedService_ServiceStruct_t * value) {
  krpc_schema_Tuple message = krpc_schema_Tuple_init_default;
  message.items.funcs.encode = &krpc_encode_callback_items_DeprecatedService_ServiceStruct;
  message.items.arg = (krpc_DeprecatedService_ServiceStruct_t*)value;
  KRPC_RETURN_ON_ERROR(krpc_encode_message_Tuple(stream, &message));
  return KRPC_OK;
}

static inline krpc_error_t krpc_encode_size_DeprecatedService_ServiceStruct(
  size_t * size, const krpc_DeprecatedService_ServiceStruct_t * value) {
  pb_ostream_t stream = PB_OSTREAM_SIZING;
  KRPC_RETURN_ON_ERROR(krpc_encode_DeprecatedService_ServiceStruct(&stream, value));
  *size = stream.bytes_written;
  return KRPC_OK;
}

static inline bool krpc_encode_callback_DeprecatedService_ServiceStruct(
  pb_ostream_t * stream, const pb_field_t * field, void * const * arg) {
  if (!pb_encode_tag_for_field(stream, field))
    KRPC_CALLBACK_RETURN_ERROR("encoding tag for DeprecatedService_ServiceStruct");
  krpc_DeprecatedService_ServiceStruct_t * value = (krpc_DeprecatedService_ServiceStruct_t*)(*arg);
  size_t size;
  KRPC_CALLBACK_RETURN_ON_ERROR(krpc_encode_size_DeprecatedService_ServiceStruct(&size, value));
  if (!pb_encode_varint(stream, size))
    KRPC_CALLBACK_RETURN_ERROR("encoding size for DeprecatedService_ServiceStruct");
  KRPC_CALLBACK_RETURN_ON_ERROR(krpc_encode_DeprecatedService_ServiceStruct(stream, value));
  return true;
}

static inline bool krpc_decode_callback_item_DeprecatedService_ServiceStruct(
  pb_istream_t * stream, const pb_field_t * field, void ** arg) {
  typedef struct State {size_t pos; krpc_DeprecatedService_ServiceStruct_t * value;} State;
  State * state = (State*)(*arg);
  krpc_DeprecatedService_ServiceStruct_t * value = state->value;
  switch (state->pos) {
  case 0:
    KRPC_CALLBACK_RETURN_ON_ERROR(krpc_decode_int32(stream, &value->field));
    break;
  default:
    // Fields are only ever appended to a structure, so a value from a newer server may
    // carry more items than this client knows about, and those are skipped
    if (!pb_read(stream, NULL, stream->bytes_left))
      KRPC_CALLBACK_RETURN_ERROR("skipping an appended struct field");
    break;
  }
  state->pos++;
  return true;
}

static inline krpc_error_t krpc_decode_DeprecatedService_ServiceStruct(
  pb_istream_t * stream, krpc_DeprecatedService_ServiceStruct_t * value) {
  krpc_schema_Tuple message = krpc_schema_Tuple_init_default;
  message.items.funcs.decode = &krpc_decode_callback_item_DeprecatedService_ServiceStruct;
  typedef struct State { size_t pos; krpc_DeprecatedService_ServiceStruct_t * value; } State;
  State state = { 0, value };
  message.items.arg = &state;
  KRPC_RETURN_ON_ERROR(krpc_decode_message_Tuple(stream, &message));
  if (state.pos < 1)
    KRPC_FAIL(ENCODING, "Too few fields for a DeprecatedService_ServiceStruct");
  return KRPC_OK;
}
