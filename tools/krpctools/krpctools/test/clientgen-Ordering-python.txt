# Aliases for the types the service declares as its attributes, as an annotation in the body of
# the service would otherwise name the attribute itself
if TYPE_CHECKING:
    pass


class ServiceA(metaclass=DeclaredClass):
    """
    A service whose procedures use another service's enumeration.
    """
    _service_name = "ServiceA"
    _class_name = None


    def frob(self, mode: _service_serviceb.Mode = _service_serviceb.Mode.fast) -> None:
        """
        Defaults to a member of another service's enumeration.
        """
        raise _declaration("ServiceA", None)

    def frob_all(self, modes: List[_service_serviceb.Mode] = [_service_serviceb.Mode.fast]) -> None:
        """
        Defaults to a collection containing a member of another service's enumeration.
        """
        raise _declaration("ServiceA", None)


class ServiceAClient(krpc.client.Client):
    """
    A client with the ServiceA service, as ``krpc.connect[ServiceAClient]()``
    connects. Combine it with the client types of other services in a subclass of them all.
    """

    _declares_service = "ServiceA"

    service_a: ServiceA


_register(
    "ServiceA",
)
