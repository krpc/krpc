class ArgumentException(ValueError):
    """
    A method was invoked where at least one of the passed arguments does not
    meet the parameter specification of the method.
    """
    _service_name = 'KRPC'
    _class_name = 'ArgumentException'

    _service_name = 'KRPC'
    _class_name = 'ArgumentNullException'

class Expression(ClassBase):
    """
    A server side expression.

    Numeric operands of differing types are converted to a common type.
    """
    @property
    def has_return_type(self) -> bool:
        """
        Whether the expression evaluates to a value.

        An expression with no value has no KRPC.Expression#return_type. An empty
        collection and no value both encode to an empty sequence of bytes.
        """
        return self._client._invoke(
            "KRPC",
            "Expression_get_HasReturnType",
            [self],
            [self._client._types.class_type("KRPC", "Expression")],
            self._client._types.bool_type
        )

    def _return_type_has_return_type(self) -> TypeBase:
        return self._client._types.bool_type

    def _build_call_has_return_type(self) -> KRPC_pb2.ProcedureCall:
        return self._client._build_call(
            "KRPC",
            "Expression_get_HasReturnType",
            [self],
            [self._client._types.class_type("KRPC", "Expression")],
            self._client._types.bool_type
        )

    @property
    def parameter_types(self) -> List[Type]:
        """
        The types of the function's parameters.

        Empty for a function that is not created with KRPC.Expression#lambda.
        """
        return self._client._invoke(
            "KRPC",
            "Expression_get_ParameterTypes",
            [self],
            [self._client._types.class_type("KRPC", "Expression")],
            self._client._types.list_type(self._client._types.class_type("KRPC", "Type"))
        )

    def _return_type_parameter_types(self) -> TypeBase:
        return self._client._types.list_type(self._client._types.class_type("KRPC", "Type"))

    def _build_call_parameter_types(self) -> KRPC_pb2.ProcedureCall:
        return self._client._build_call(
            "KRPC",
            "Expression_get_ParameterTypes",
            [self],
            [self._client._types.class_type("KRPC", "Expression")],
            self._client._types.list_type(self._client._types.class_type("KRPC", "Type"))
        )

    @property
    def return_type(self) -> Type:
        """
        The type of the value the expression evaluates to. Throws an exception if the
        value cannot be sent to a client.

        A function created with KRPC.Expression#lambda evaluates to the value it returns. A
        lazily evaluated collection, such as from KRPC.Expression#select, is sent once
        converted by KRPC.Expression#to_list or KRPC.Expression#to_set.
        """
        return self._client._invoke(
            "KRPC",
            "Expression_get_ReturnType",
            [self],
            [self._client._types.class_type("KRPC", "Expression")],
            self._client._types.class_type("KRPC", "Type")
        )

    def _return_type_return_type(self) -> TypeBase:
        return self._client._types.class_type("KRPC", "Type")

    def _build_call_return_type(self) -> KRPC_pb2.ProcedureCall:
        return self._client._build_call(
            "KRPC",
            "Expression_get_ReturnType",
            [self],
            [self._client._types.class_type("KRPC", "Expression")],
            self._client._types.class_type("KRPC", "Type")
        )

    @StaticMethod
    def call(cls, call: KRPC_pb2.ProcedureCall) -> Expression:
        """
        An RPC call.

        The instance the call is made on, and the values of its arguments, are fixed
        when the expression is created. A call to a procedure with no return value is
        evaluated for its effects, as a statement in a KRPC.Expression#block. A collection the
        procedure returns is copied, as a client would receive it.

        :param call: The procedure call.

    def __init__(self, client: Client) -> None:
        self._client = client
        # The classes, enumerations, structures and exceptions the service defines are
        # put on the service here, so that reaching one is an ordinary attribute lookup
        # and nothing has to intercept the lookups that do not name one - which is every
        # call, as a call goes through a procedure or a property. A class is bound to the
        # client it is reached through, so that its static methods use that client.
        self.__dict__.update(self._enumerations)
        self.__dict__.update(self._structs)
        self.__dict__.update(self._exceptions)
        for name, class_type in self._classes.items():
            self.__dict__[name] = WrappedClass(client, class_type)

    # Hidden from type checkers, so that they report a name the service does not define
    if not TYPE_CHECKING:

        def __getattr__(self, name):
            # Reached only once the ordinary lookup has failed, which for a service means
            # either a name it does not define, or one it does whose procedure or property
            # raised an AttributeError of its own. Python discards that error before calling
            # this, so it cannot be reported here; what can be done is to not report a member
            # that does exist as missing, which would point at the wrong thing entirely.
            cls = type(self)
            if hasattr(cls, name):
                raise AttributeError(
                    "'%s' object attribute '%s' raised AttributeError"
                    % (cls.__name__, name)
                )
            raise AttributeError(
                "'%s' object has no attribute '%s'" % (cls.__name__, name)
            )

    def hold_tick(self) -> None:
        """
        Hold the game on its current physics tick until KRPC#release_tick is called
        or the hold times out. Every call made in between is executed before the game
        advances, so a read, compute and write all happen in the state of one tick.

        The game renders no frame, takes no input and runs no physics while the tick is
        held. Keep a hold as short as possible, and release it even when the code between
        the two calls fails.

        A tick is held once, so this waits for the next tick when the current one has
        already been released. One client holds the tick at a time, and only a client
        making a call can hold it: a stream or an event cannot.

        A hold ends by itself when it times out, when the client disconnects, and when the
        client calls a procedure that takes more than one tick to finish, such as staging
        or warping. Stream updates are sent once the tick is released, so waiting for a
        stream or an event inside a hold waits out the timeout.
        """
        return self._client._invoke(
            "KRPC",
            "HoldTick",
            [],
            [],
            None
        )

    def _return_type_hold_tick(self) -> TypeBase:
        return None

    def _build_call_hold_tick(self) -> KRPC_pb2.ProcedureCall:
        return self._client._build_call(
            "KRPC",
            "HoldTick",
            [],
            [],
            None
        )

    def next_tick(self) -> None:
        """
        Let the game move on from the tick being held and hold the one after it, so that a
        program acts on consecutive physics ticks. Holds the soonest tick it can if no tick
        is currently being held.

        A release and a hold made as two separate calls let the game advance between them.
        This takes the next tick without that gap. Call it at the top of a loop that has to
        act on every tick, work, and come back to it.

        The game advances one tick per call, so it runs at the rate of the loop for as long
        as the loop runs. End the loop with KRPC#release_tick.

        Every limit of KRPC#hold_tick applies, the timeout included. A hold lost to
        the timeout is not reported, and the next call takes whatever tick it lands on.
        """
        return self._client._invoke(
            "KRPC",
            "NextTick",
            [],
            [],
            None
        )

        return None
