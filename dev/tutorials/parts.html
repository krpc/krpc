Interacting with Parts#

The following examples use the parts of a vessel to achieve various tasks. The parts guide explains the tree of parts, attachment modes, fuel lines, staging and part modules.

Deploying all Parachutes#

Sometimes things go horribly wrong. The following script does its best to save your Kerbals by deploying all the parachutes:

import krpc

conn = krpc.connect()
vessel = conn.space_center.active_vessel

for parachute in vessel.parts.parachutes:
    parachute.deploy()

using KRPC.Client;
using KRPC.Client.Services.SpaceCenter;

class DeployParachutes
{
    public static void Main ()
    {
        using (var connection = new Connection ()) {
            var vessel = connection.SpaceCenter ().ActiveVessel;
            foreach (var parachute in vessel.Parts.Parachutes)
                parachute.Deploy ();
        }
    }
}

#include <krpc.hpp>
#include <krpc/services/space_center.hpp>

int main() {
  krpc::Connection conn = krpc::connect();
  krpc::services::SpaceCenter space_center(conn);
  auto vessel = space_center.active_vessel().value();
  for (auto parachute : vessel.parts().parachutes()) parachute.deploy();
}

#include <krpc_cnano.h>
#include <krpc_cnano/services/space_center.h>

int main() {
  krpc_connection_t conn;
  if (krpc_open_serial(&conn, "COM0") != KRPC_OK) return 1;
  if (krpc_connect(conn, "Deploy parachutes") != KRPC_OK) return 1;

  krpc_SpaceCenter_Vessel_t vessel;
  krpc_SpaceCenter_ActiveVessel(conn, &vessel);

  krpc_SpaceCenter_Parts_t parts;
  krpc_SpaceCenter_Vessel_Parts(conn, &parts, vessel);

  krpc_list_object_t parachutes = KRPC_NULL_LIST;
  krpc_SpaceCenter_Parts_Parachutes(conn, &parachutes, parts);
  for (size_t i = 0; i < parachutes.size; i++) {
    krpc_SpaceCenter_Parachute_t parachute = parachutes.items[i];
    krpc_SpaceCenter_Parachute_Deploy(conn, parachute);
  }
  KRPC_FREE_LIST(parachutes);

  krpc_disconnect(conn);
}

The script deploys the parachutes one call at a time, and the game can advance between the calls. A server side function deploys them all in the same physics tick, with a single call. Python and JavaScript compile it from a function. A C# lambda holds no loop, so C# and the other clients build the loop with the expression API:

def deploy_parachutes():
    for parachute in vessel.parts.parachutes:
        parachute.deploy()


conn.run(deploy_parachutes)

    "parachute", t.ClassType ("SpaceCenter", "Parachute"));
var deploy = e.CallWithArguments (
    Connection.GetCallTemplate ((Parachute p) => p.Deploy ()),
    new Dictionary<int, Expression> { { 0, parachute } });

    new List<Expression> { parachute },

‘Control From Here’ for Docking Ports#

The following example will find a standard sized Clamp-O-Tron docking port, and control the vessel from it:

import krpc

conn = krpc.connect()
vessel = conn.space_center.active_vessel
part = vessel.parts.with_title("Clamp-O-Tron Docking Port")[0]
vessel.parts.controlling = part

using KRPC.Client;
using KRPC.Client.Services.SpaceCenter;

class ControlFromHere
{
    public static void Main ()
    {
        using (var conn = new Connection ()) {
            var vessel = conn.SpaceCenter ().ActiveVessel;
            var part =
                vessel.Parts.WithTitle ("Clamp-O-Tron Docking Port") [0];
            vessel.Parts.Controlling = part;
        }
    }
}

#include <krpc.hpp>
#include <krpc/services/space_center.hpp>

int main() {
  krpc::Connection conn = krpc::connect();
  krpc::services::SpaceCenter space_center(conn);
  auto vessel = space_center.active_vessel().value();
  auto part =
      vessel.parts().with_title("Clamp-O-Tron Docking Port").front();
  vessel.parts().set_controlling(part);
}

#include <krpc_cnano.h>
#include <krpc_cnano/services/space_center.h>

int main() {
  krpc_connection_t conn;
  if (krpc_open_serial(&conn, "COM0") != KRPC_OK) return 1;
  if (krpc_connect(conn, "Control from here") != KRPC_OK) return 1;

  krpc_SpaceCenter_Vessel_t vessel;
  krpc_SpaceCenter_ActiveVessel(conn, &vessel);

  krpc_SpaceCenter_Parts_t parts;
  krpc_SpaceCenter_Vessel_Parts(conn, &parts, vessel);

  krpc_list_object_t docking_port_parts = KRPC_NULL_LIST;
  krpc_SpaceCenter_Parts_WithTitle(conn, &docking_port_parts, parts,
                                   "Clamp-O-Tron Docking Port");
  if (docking_port_parts.size == 0) {
    KRPC_FREE_LIST(docking_port_parts);
    krpc_disconnect(conn);
    return 1;
  }
  krpc_object_t part = docking_port_parts.items[0];
  KRPC_FREE_LIST(docking_port_parts);
  krpc_SpaceCenter_Parts_set_Controlling(conn, parts, part);

  krpc_disconnect(conn);
}

Combined Specific Impulse#

The following script calculates the combined specific impulse of all currently active and fueled engines on a rocket. See here for a description of the maths: https://wiki.kerbalspaceprogram.com/wiki/Specific_impulse#Multiple_engines

import krpc

conn = krpc.connect()
vessel = conn.space_center.active_vessel

active_engines = [
    e for e in vessel.parts.engines if e.active and e.has_fuel
]

print("Active engines:")
for engine in active_engines:
    print("   %s in stage %d" % (engine.part.title, engine.part.stage))

thrust = sum(engine.available_thrust for engine in active_engines)
fuel_consumption = sum(
    engine.available_thrust / engine.specific_impulse
    for engine in active_engines
)
isp = thrust / fuel_consumption

print("Combined Isp = %.0f seconds" % isp)

using System;
using System.Linq;
using KRPC.Client;
using KRPC.Client.Services.SpaceCenter;

class CombinedIsp
{
    public static void Main ()
    {
        using (var connection = new Connection ()) {
            var vessel = connection.SpaceCenter ().ActiveVessel;

            var activeEngines = vessel.Parts.Engines
                .Where (e => e.Active && e.HasFuel).ToList ();

            Console.WriteLine ("Active engines:");
            foreach (var engine in activeEngines)
                Console.WriteLine ("   " + engine.Part.Title +
                                   " in stage " + engine.Part.Stage);

            double thrust = activeEngines.Sum (e => e.AvailableThrust);
            double fuel_consumption =
                activeEngines.Sum (
                    e => e.AvailableThrust / e.SpecificImpulse);
            double isp = thrust / fuel_consumption;
            Console.WriteLine ("Combined Isp = {0:F0} seconds", isp);
        }
    }
}

#include <iomanip>
#include <iostream>
#include <krpc.hpp>
#include <krpc/services/space_center.hpp>
#include <vector>

using SpaceCenter = krpc::services::SpaceCenter;

int main() {
  auto conn = krpc::connect();
  SpaceCenter sc(conn);
  auto vessel = sc.active_vessel().value();

  auto engines = vessel.parts().engines();

  std::vector<SpaceCenter::Engine> active_engines;
  for (auto engine : engines)
    if (engine.active() && engine.has_fuel())
      active_engines.push_back(engine);

  std::cout << "Active engines:" << std::endl;
  for (auto engine : active_engines)
    std::cout << "   " << engine.part().title() << " in stage "
              << engine.part().stage() << std::endl;

  double thrust = 0;
  double fuel_consumption = 0;
  for (auto engine : active_engines) {
    thrust += engine.available_thrust();
    fuel_consumption +=
        engine.available_thrust() / engine.specific_impulse();
  }
  double isp = thrust / fuel_consumption;
  std::cout << "Combined Isp = " << std::fixed << std::setprecision(0)
            << isp << " seconds" << std::endl;
}

#include <krpc_cnano.h>
#include <krpc_cnano/services/space_center.h>
#include <stdint.h>
#include <stdio.h>

int main() {
  krpc_connection_t conn;
  if (krpc_open_serial(&conn, "COM0") != KRPC_OK) return 1;
  if (krpc_connect(conn, "Combined Isp") != KRPC_OK) return 1;

  krpc_SpaceCenter_Vessel_t vessel;
  krpc_SpaceCenter_ActiveVessel(conn, &vessel);

  krpc_SpaceCenter_Parts_t parts;
  krpc_SpaceCenter_Vessel_Parts(conn, &parts, vessel);

  krpc_list_object_t engines = KRPC_NULL_LIST;
  krpc_SpaceCenter_Parts_Engines(conn, &engines, parts);

  krpc_list_object_t active_engines = KRPC_NULL_LIST;
  active_engines.size = 0;
  active_engines.items = krpc_calloc(engines.size, sizeof(krpc_object_t));

  for (size_t i = 0; i < engines.size; i++) {
    krpc_SpaceCenter_Engine_t engine = engines.items[i];
    bool active;
    bool has_fuel;
    krpc_SpaceCenter_Engine_Active(conn, &active, engine);
    krpc_SpaceCenter_Engine_HasFuel(conn, &has_fuel, engine);
    if (active && has_fuel) {
      active_engines.items[active_engines.size] = engine;
      active_engines.size++;
    }
  }

  printf("Active engines:\n");
  for (size_t i = 0; i < active_engines.size; i++) {
    krpc_SpaceCenter_Engine_t engine = active_engines.items[i];
    krpc_SpaceCenter_Part_t part;
    krpc_SpaceCenter_Engine_Part(conn, &part, engine);
    char* title = NULL;
    int32_t stage;
    krpc_SpaceCenter_Part_Title(conn, &title, part);
    krpc_SpaceCenter_Part_Stage(conn, &stage, part);
    printf("   %s in stage %ld\n", title, (long)stage);
    krpc_free(title);
  }

  double thrust = 0;
  double fuel_consumption = 0;
  for (size_t i = 0; i < active_engines.size; i++) {
    krpc_SpaceCenter_Engine_t engine = active_engines.items[i];
    float engine_thrust;
    float engine_isp;
    krpc_SpaceCenter_Engine_AvailableThrust(conn, &engine_thrust, engine);
    krpc_SpaceCenter_Engine_SpecificImpulse(conn, &engine_isp, engine);
    thrust += engine_thrust;
    fuel_consumption += engine_thrust / engine_isp;
  }
  double isp = thrust / fuel_consumption;
  printf("Combined Isp = %.0f seconds\n", isp);

  KRPC_FREE_LIST(engines);
  KRPC_FREE_LIST(active_engines);

  krpc_disconnect(conn);
}

The script reads the thrust and specific impulse of each engine with separate calls, which can land in different physics ticks. A server side function reads them all in the same tick, with a single call. Python, C# and JavaScript compile it from a function, and the other clients build it with the expression API:

def combined_isp():
    engines = [e for e in vessel.parts.engines if e.active and e.has_fuel]
    thrust = sum(engine.available_thrust for engine in engines)
    fuel_consumption = sum(
        engine.available_thrust / engine.specific_impulse
        for engine in engines
    )
    return thrust / fuel_consumption


isp = conn.run(combined_isp)

var isp = connection.Run (() =>
    vessel.Parts.Engines.Where (e => e.Active && e.HasFuel)
        .Select (e => e.AvailableThrust).Sum () /
    vessel.Parts.Engines.Where (e => e.Active && e.HasFuel)
        .Select (e => e.AvailableThrust / e.SpecificImpulse)
        .Sum ());

auto e = Expr::bind(conn);
auto t = KType::bind(conn);
auto engine =
    e.parameter("engine", t.class_type("SpaceCenter", "Engine"));
// A property of the engine parameter
auto get = [&](const krpc::schema::ProcedureCall& property) {
  return e.call_with_arguments(property,
                               std::map<int32_t, Expr>({{0, engine}}));
};
// A function of the engine parameter
auto function = [&](const Expr& body) {
  return e.lambda(std::vector<Expr>({engine}), body);
};

// The active engines with fuel
auto engines = e.where(
    e.call(vessel.parts().engines_call()),
    function(e.conditional_and(get(Engine::active_call_template()),
                               get(Engine::has_fuel_call_template()))));
auto thrust = get(Engine::available_thrust_call_template());
auto isp = get(Engine::specific_impulse_call_template());
auto total_thrust = e.sum(e.select(engines, function(thrust)));
auto fuel_consumption =
    e.sum(e.select(engines, function(e.divide(thrust, isp))));

auto combined_isp =
    krpc::run<float>(e.divide(total_thrust, fuel_consumption));

typedef krpc_error_t (*call_template_t)(krpc_schema_ProcedureCall*);

// A property of the engine parameter
static void get(krpc_connection_t conn, krpc_KRPC_Expression_t* result,
                krpc_KRPC_Expression_t engine, call_template_t property) {
  krpc_schema_ProcedureCall call;
  property(&call);
  krpc_dictionary_entry_int32_object_t entries[1] = {{0, engine}};
  krpc_dictionary_int32_object_t arguments = {1, entries};
  krpc_KRPC_Expression_CallWithArguments(conn, result, &call, &arguments);
  krpc_free_procedure_call(&call);
}

// A function of the engine parameter
static void function(krpc_connection_t conn, krpc_KRPC_Expression_t* result,
                     krpc_KRPC_Expression_t engine,
                     krpc_KRPC_Expression_t body) {
  krpc_list_object_t parameters = {1, &engine};
  krpc_KRPC_Expression_Lambda(conn, result, &parameters, body);
}

  return Expression.lambda({engine}, body)

-- The active engines with fuel
local engines = Expression.where(
  Expression.call(conn:get_call(parts.get_engines, parts)),
  fn(Expression.conditional_and(
    get(Engine.get_active), get(Engine.get_has_fuel))))
local thrust = get(Engine.get_available_thrust)
local isp = get(Engine.get_specific_impulse)
local total_thrust = Expression.sum(Expression.select(engines, fn(thrust)))
local fuel_consumption = Expression.sum(
  Expression.select(engines, fn(Expression.divide(thrust, isp))))

  conn:run(Expression.divide(total_thrust, fuel_consumption))

const isp = await conn.run(
  ({ vessel }) => {
    let thrust = 0;
    let fuelConsumption = 0;
    for (const engine of vessel.parts.engines) {
      if (engine.active && engine.hasFuel) {
        thrust += engine.availableThrust;
        fuelConsumption += engine.availableThrust / engine.specificImpulse;
      }
    }
    return thrust / fuelConsumption;
  },
