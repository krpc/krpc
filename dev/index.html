kRPC - Remote Procedure Calls for Kerbal Space Program#

kRPC lets you control Kerbal Space Program from scripts running outside the game. A server inside the game exposes the game’s API, and client libraries call it from Python, C#, C++, C, Java and Lua. It also supports Ferram Aerospace Research, Kerbal Alarm Clock, Infernal Robotics and other popular mods.

This script launches the active vessel straight up, then cuts the engines once it reaches an altitude of 10 km:

import krpc

conn = krpc.connect(name="Launch")
vessel = conn.space_center.active_vessel
flight = vessel.flight()

vessel.auto_pilot.target_pitch_and_heading(90, 90)
vessel.auto_pilot.engaged = True
vessel.control.throttle = 1
vessel.control.activate_next_stage()
print(f"{vessel.name} has launched")

event = conn.add_event(lambda: flight.mean_altitude > 10000)
with event.condition:
    event.wait()

vessel.control.throttle = 0
vessel.auto_pilot.engaged = False
print(f"{vessel.name} reached 10 km")

using System;
using KRPC.Client;
using KRPC.Client.Services.SpaceCenter;

class Launch
{
    public static void Main ()
    {
        var conn = new Connection ("Launch");
        var vessel = conn.SpaceCenter ().ActiveVessel;
        var flight = vessel.Flight ();

        vessel.AutoPilot.TargetPitchAndHeading (90, 90);
        vessel.AutoPilot.Engaged = true;
        vessel.Control.Throttle = 1;
        vessel.Control.ActivateNextStage ();
        Console.WriteLine (vessel.Name + " has launched");

        var evnt = conn.AddEvent (() => flight.MeanAltitude > 10000);
        lock (evnt.Condition) {
            evnt.Wait ();
        }

        vessel.Control.Throttle = 0;
        vessel.AutoPilot.Engaged = false;
        Console.WriteLine (vessel.Name + " reached 10 km");
    }
}

#include <iostream>
#include <krpc.hpp>
#include <krpc/function_stream.hpp>
#include <krpc/services/krpc.hpp>
#include <krpc/services/space_center.hpp>

int main() {
  krpc::Client conn = krpc::connect("Launch");
  krpc::services::SpaceCenter space_center(&conn);
  auto vessel = space_center.active_vessel().value();
  auto flight = vessel.flight();

  vessel.auto_pilot().target_pitch_and_heading(90, 90);
  vessel.auto_pilot().set_engaged(true);
  vessel.control().set_throttle(1);
  vessel.control().activate_next_stage();
  std::cout << vessel.name() << " has launched" << std::endl;

  typedef krpc::services::KRPC::Expression Expr;
  auto expr = Expr::greater_than(
    conn, Expr::call(conn, flight.mean_altitude_call()), Expr::constant_double(conn, 10000));
  auto event = krpc::add_event(expr);
  event.acquire();
  event.wait();
  event.release();

  vessel.control().set_throttle(0);
  vessel.auto_pilot().set_engaged(false);
  std::cout << vessel.name() << " reached 10 km" << std::endl;
}

#include <stdio.h>
#include <stdlib.h>
#include <krpc_cnano.h>
#include <krpc_cnano/services/space_center.h>

int main() {
  krpc_connection_t conn;
  krpc_open(&conn, "COM0");
  krpc_connect(conn, "Launch");

  krpc_SpaceCenter_Vessel_t vessel;
  krpc_SpaceCenter_ActiveVessel(conn, &vessel);
  krpc_SpaceCenter_AutoPilot_t auto_pilot;
  krpc_SpaceCenter_Vessel_AutoPilot(conn, &auto_pilot, vessel);
  krpc_SpaceCenter_Control_t control;
  krpc_SpaceCenter_Vessel_Control(conn, &control, vessel);
  krpc_SpaceCenter_Flight_t flight;
  krpc_SpaceCenter_Vessel_Flight(conn, &flight, vessel, KRPC_NULL);
  char * name = NULL;
  krpc_SpaceCenter_Vessel_Name(conn, &name, vessel);

  krpc_SpaceCenter_AutoPilot_TargetPitchAndHeading(conn, auto_pilot, 90, 90);
  krpc_SpaceCenter_AutoPilot_set_Engaged(conn, auto_pilot, true);
  krpc_SpaceCenter_Control_set_Throttle(conn, control, 1);
  krpc_SpaceCenter_Control_ActivateNextStage(conn, NULL, control);
  printf("%s has launched\n", name);

  double mean_altitude = 0;
  while (mean_altitude < 10000)
    krpc_SpaceCenter_Flight_MeanAltitude(conn, &mean_altitude, flight);

  krpc_SpaceCenter_Control_set_Throttle(conn, control, 0);
  krpc_SpaceCenter_AutoPilot_set_Engaged(conn, auto_pilot, false);
  printf("%s reached 10 km\n", name);
  free(name);
}

import krpc.client.Connection;
import krpc.client.Event;
import krpc.client.RPCException;
import krpc.client.StreamException;
import krpc.client.services.KRPC.Expression;
import krpc.client.services.SpaceCenter;
import krpc.schema.KRPC.ProcedureCall;

import java.io.IOException;

public class Launch {
  public static void main(String[] args) throws IOException, RPCException, StreamException {
    try (Connection connection = Connection.newInstance("Launch")) {
      SpaceCenter spaceCenter = SpaceCenter.newInstance(connection);
      SpaceCenter.Vessel vessel = spaceCenter.getActiveVessel();
      SpaceCenter.Flight flight = vessel.flight(null);

      vessel.getAutoPilot().targetPitchAndHeading(90, 90);
      vessel.getAutoPilot().setEngaged(true);
      vessel.getControl().setThrottle(1);
      vessel.getControl().activateNextStage();
      System.out.println(vessel.getName() + " has launched");

      ProcedureCall meanAltitude = connection.getCall(flight, "getMeanAltitude");
      Expression expr = Expression.greaterThan(
        connection,
        Expression.call(connection, meanAltitude),
        Expression.constantDouble(connection, 10000));
      Event event = connection.addEvent(expr);
      synchronized (event.getCondition()) {
        event.waitFor();
      }

      vessel.getControl().setThrottle(0);
      vessel.getAutoPilot().setEngaged(false);
      System.out.println(vessel.getName() + " reached 10 km");
    }
  }
}

local krpc = require 'krpc'
local platform = require 'krpc.platform'
local conn = krpc.connect('Launch')
local vessel = conn.space_center.active_vessel
local flight = vessel:flight()

vessel.auto_pilot:target_pitch_and_heading(90, 90)
vessel.auto_pilot.engaged = true
vessel.control.throttle = 1
vessel.control:activate_next_stage()
print(vessel.name .. ' has launched')

while flight.mean_altitude < 10000 do
    platform.sleep(0.1)
end

vessel.control.throttle = 0
vessel.auto_pilot.engaged = false
print(vessel.name .. ' reached 10 km')
